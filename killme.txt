ProtoFlow database

users: 
 - id BIGINT SERIAL PK
 - tag VARCHAR(150)

 - name VARCHAR(150)
 - phone VARCHAR(20)
 - email VARCHAR(150) UNIQUE
 - password_hash TEXT

 - avatar_url_id BIGINT FK
 
 - document_type VARCHAR(20)
 - document_number VARCHAR(20)
 - timezone VARCHAR(50)

 - note TEXT
 - status ENUM('active','inactive') 
 
 
 - last_login_at TIMESTAMP
 - email_verified_at TIMESTAMP
 - terms_accepted_at TIMESTAMP
 - created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
 - updated_at TIMESTAMP


roles: 
 - id INT SERIAL PK

 - name VARCHAR(255)

 - status ENUM('active','inactive') 
 
 - created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
 - updated_at TIMESTAMP

user_roles:
 - id BIGINT SERIAL PK
 - user_id BIGINT FK
 - role_id BIGINT FK
 
 - status ENUM('active','inactive') 
 
 - created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
 - updated_at TIMESTAMP

file_types:
 - id INT pk

 - name VARCHAR(20)
 
 - status ENUM('active','inactive') 
 
 - created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
 - updated_at TIMESTAMP
 

files
 - id BIGINT PK
 - type_id INT FK
 
 - filename VARCHAR(255)
 - path VARCHAR(500)
 - mime_type VARCHAR(100)
 - size_bytes BIGINT
 
 - status ENUM('active','inactive') 

 - created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
 - updated_at TIMESTAMP
